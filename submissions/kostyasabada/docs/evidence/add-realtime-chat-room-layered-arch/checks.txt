#!/usr/bin/env bash
# Runs the planning checks for add-realtime-chat-room-layered-arch and prints each
# command, its output, and its real exit code.

cd /home/ksabada/projects/AI_course/2026-agentic-engineering-crash-course-capstone/submissions/kostyasabada
export PATH=/tmp/node-v24.21.0-linux-x64/bin:$PATH
run() { echo "\$ $*"; "$@" 2>&1; echo "exit=$?"; echo; }
echo "date: $(date -u +%Y-%m-%dT%H:%M:%SZ)"
echo "cwd: submissions/kostyasabada"
echo
run node --version
run npm run --silent openspec -- validate add-realtime-chat-room --strict
run npm run --silent openspec -- validate --all --strict
echo '$ npm run --silent openspec -- status --change add-realtime-chat-room --json | python3 -c <summary of artifact statuses>'
npm run --silent openspec -- status --change add-realtime-chat-room --json 2>&1 | python3 -c 'import json,sys; d=json.load(sys.stdin); print("isPlanningComplete=",d["isPlanningComplete"]); [print(a["id"],a["status"]) for a in d["artifacts"]]'
echo "exit=${PIPESTATUS[0]}"; echo
run git diff --check
D=docs/evidence/add-realtime-chat-room-layered-arch
echo "\$ grep -nE '[[:space:]]\$' $D/implementation.md"
grep -nE '[[:space:]]$' "$D/implementation.md"; echo "exit=$? (1 = no trailing whitespace found)"; echo
echo "\$ grep -rn 'message-store\\|socket-handlers\\|src/server/host-policy' openspec/changes/add-realtime-chat-room/tasks.md openspec/changes/add-realtime-chat-room/specs openspec/changes/add-realtime-chat-room/proposal.md"

run git status --short
