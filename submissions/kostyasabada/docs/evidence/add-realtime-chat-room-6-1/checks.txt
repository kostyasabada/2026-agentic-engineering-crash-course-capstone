## Helper body: paths.py
# Lists every backticked token in the given docs that looks like a repo path
# and reports whether it exists (relative to the submission root, or to the
# document's own directory for relative links).
import os, re, sys
root = os.getcwd()
for doc in sys.argv[1:]:
    text = open(doc, encoding='utf-8').read()
    seen = set()
    for tok in re.findall(r'`([^`\n]+)`', text):
        t = tok.strip()
        if ' ' in t or t.startswith(('http', 'npm', 'node', 'npx', '-', '<', '~', '/tmp')):
            continue
        if not ('/' in t or re.search(r'\.(ts|tsx|md|mjs|json|sh|css|txt|log|yaml)$', t)):
            continue
        if '<' in t or '*' in t or t in seen:
            continue
        seen.add(t)
        cands = [os.path.join(root, t), os.path.join(os.path.dirname(doc), t)]
        ok = any(os.path.exists(c) for c in cands)
        print(f"{'OK     ' if ok else 'MISSING'} {doc}: {t}")

### $SP/phaseb/two-browsers.mjs
// two-browsers.mjs (task 6.1 phase B, SUPPLEMENTARY manual session; not a test). Drives the
// already running `npm start` server at BASE with two independent headless Chromium contexts
// (Alice, Bob), exchanges messages, prints what each context shows, and saves screenshots:
// each context separately, and both side by side (composited in a third page from the two PNGs).
import path from 'node:path'
import { readFile } from 'node:fs/promises'
import { createRequire } from 'node:module'
const SUB = process.env.SUB, OUT = process.env.SHOTS, BASE = process.env.BASE
const { chromium, expect } = createRequire(path.join(SUB, 'package.json'))('@playwright/test')
const browser = await chromium.launch()
try {
  const open = async (nickname) => {
    const context = await browser.newContext({ baseURL: BASE, viewport: { width: 760, height: 620 } })
    const page = await context.newPage()
    page.on('pageerror', (e) => console.log(`${nickname} pageerror: ${e.message}`))
    page.on('console', (m) => { if (m.type() === 'error') console.log(`${nickname} console error: ${m.text()}`) })
    await page.goto('/')
    await page.getByLabel('Nickname', { exact: true }).fill(nickname)
    await page.getByRole('button', { name: 'Join' }).click()
    await expect(page.getByText(/^Chatting as /)).toHaveText(`Chatting as ${nickname}`)
    await expect(page.getByRole('button', { name: 'Send', exact: true })).toBeEnabled()
    return page
  }
  const send = async (page, text) => {
    const input = page.getByRole('textbox', { name: 'Message', exact: true })
    await input.fill(text)
    await input.press('Enter')
    await expect(input).toHaveValue('')
  }
  const shown = (page) => page.getByRole('list', { name: 'Messages' }).getByRole('listitem').evaluateAll((items) =>
    items.map((i) => `${i.querySelector('.message__author')?.textContent}: ${i.querySelector('.message__text')?.textContent}`))
  const status = (page) => page.getByRole('region', { name: 'Chat', exact: true }).getByRole('status').textContent()
  const alice = await open('Alice')
  const bob = await open('Bob')
  await send(alice, 'Hi Bob, this is Alice (phase B manual session).')
  await send(bob, 'Hi Alice, Bob here. Your message arrived.')
  await send(alice, 'Great, real-time delivery works in both directions.')
  const expected = [
    'Alice: Hi Bob, this is Alice (phase B manual session).',
    'Bob: Hi Alice, Bob here. Your message arrived.',
    'Alice: Great, real-time delivery works in both directions.',
  ]
  for (const [name, page] of [['Alice', alice], ['Bob', bob]]) {
    await expect.poll(() => shown(page)).toEqual(expected)
    console.log(`${name} status: ${await status(page)}`)
    console.log(`${name} shows:\n  ${(await shown(page)).join('\n  ')}`)
  }
  const a = path.join(OUT, 'supplementary-alice.png'), b = path.join(OUT, 'supplementary-bob.png')
  await alice.screenshot({ path: a })
  await bob.screenshot({ path: b })
  const uri = async (f) => `data:image/png;base64,${(await readFile(f)).toString('base64')}`
  const combo = await (await browser.newContext({ viewport: { width: 1560, height: 680 } })).newPage()
  await combo.setContent(`<body style="margin:0;padding:10px;background:#888;display:flex;gap:20px;font:14px sans-serif">
    <figure style="margin:0"><img src="${await uri(a)}"><figcaption>Context 1: Alice (supplementary)</figcaption></figure>
    <figure style="margin:0"><img src="${await uri(b)}"><figcaption>Context 2: Bob (supplementary)</figcaption></figure></body>`)
  await combo.screenshot({ path: path.join(OUT, 'supplementary-two-browsers.png'), fullPage: true })
  console.log('screenshots saved')
} finally {
  await browser.close()
}
