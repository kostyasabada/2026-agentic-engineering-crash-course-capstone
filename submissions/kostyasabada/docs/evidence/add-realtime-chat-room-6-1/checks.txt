## Helper body: paths.py
# Lists every backticked token in the given docs that looks like a repo path
# and reports whether it exists (relative to the submission root, or to the
# document's own directory for relative links).
import os, re, sys
root = os.getcwd()
for doc in sys.argv[1:]:
    text = open(doc, encoding='utf-8').read()
    seen = set()
    for tok in re.findall(r'`([^`\n]+)`', text):
        t = tok.strip()
        if ' ' in t or t.startswith(('http', 'npm', 'node', 'npx', '-', '<', '~', '/tmp')):
            continue
        if not ('/' in t or re.search(r'\.(ts|tsx|md|mjs|json|sh|css|txt|log|yaml)$', t)):
            continue
        if '<' in t or '*' in t or t in seen:
            continue
        seen.add(t)
        cands = [os.path.join(root, t), os.path.join(os.path.dirname(doc), t)]
        ok = any(os.path.exists(c) for c in cands)
        print(f"{'OK     ' if ok else 'MISSING'} {doc}: {t}")
