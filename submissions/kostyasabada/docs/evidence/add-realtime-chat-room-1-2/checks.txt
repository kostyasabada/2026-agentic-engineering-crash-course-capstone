// Run from submissions/kostyasabada/: node <tsx/cli> <this file>  (NODE_ENV=production build needed)
// Usage: ... forced-kill-check.mts grandchild|cli
// Starts a server with the real ChatServerProcess, freezes one process with SIGSTOP (the real
// server.ts process = "grandchild", or the tsx CLI = "cli") so that SIGTERM cannot end it,
// calls stop() (SIGTERM, then SIGKILL after 5 s), and reports which server processes remain.
import { execFileSync } from 'node:child_process'
import path from 'node:path'
import { mkdtempSync } from 'node:fs'
import { tmpdir } from 'node:os'
import { ChatServerProcess } from '/home/ksabada/projects/AI_course/2026-agentic-engineering-crash-course-capstone/submissions/kostyasabada/e2e/fixtures/chat-server.ts'

function servers(): string {
  try {
    return execFileSync('pgrep', ['-af', '^/?[^ ]*node .*server\\.ts$'], { encoding: 'utf8' }).trim()
  } catch {
    return '(none)'
  }
}

const port = 3571

console.log('server processes after start():\n' + servers())
const target = process.argv[2]
const pattern = target === 'cli' ? 'tsx/dist/cli\\.mjs server\\.ts$' : '--import .*tsx.*server\\.ts$'
const frozen = execFileSync('pgrep', ['-f', '--', pattern], { encoding: 'utf8' }).trim()
console.log(`frozen with SIGSTOP (${target}):`, frozen)
process.kill(Number(frozen), 'SIGSTOP')
const t0 = Date.now()
await server.stop()
console.log(`stop() returned after ${Date.now() - t0} ms`)
await new Promise((r) => setTimeout(r, 500))
const left = servers()
console.log('server processes after stop():\n' + left)
if (left !== '(none)') {
  // Clean up so that the orphan does not keep the port.
  for (const line of left.split('\n')) process.kill(Number(line.split(' ')[0]), 'SIGKILL')

  process.exit(1)
}
