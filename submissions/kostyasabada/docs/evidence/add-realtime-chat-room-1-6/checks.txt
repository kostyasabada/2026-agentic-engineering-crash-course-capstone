#!/usr/bin/env bash
# serve.sh <npm script>: runs `npm run <script>` (as documented in README.md) in its own process
# group, waits up to 60 s for the "> Ready on" line, requests / with curl, sends SIGINT (Ctrl+C)
# to the whole group, and reports leftover processes and the port state.
set -u
script=$1
log=$(mktemp)
setsid npm run "$script" >"$log" 2>&1 &
pgid=$!
for _ in $(seq 1 120); do grep -q '> Ready on' "$log" && break; sleep 0.5; done
cat "$log"
echo "--- curl -sI http://127.0.0.1:3000/"
curl -sI --max-time 20 http://127.0.0.1:3000/ | head -n 1
echo "curl exit=${PIPESTATUS[0]}"
kill -INT -- "-$pgid" 2>/dev/null
for _ in $(seq 1 20); do kill -0 -- "-$pgid" 2>/dev/null || break; sleep 0.5; done
if kill -0 -- "-$pgid" 2>/dev/null; then echo "group still alive after SIGINT; sending SIGKILL"; kill -KILL -- "-$pgid"; fi
wait "$pgid" 2>/dev/null; echo "npm run $script exit after SIGINT=$?"
pgrep -g "$pgid" -a || echo "no processes left in group $pgid"
ss -ltn | grep -q ':3000 ' && echo 'port 3000 still listening' || echo 'port 3000 free'
