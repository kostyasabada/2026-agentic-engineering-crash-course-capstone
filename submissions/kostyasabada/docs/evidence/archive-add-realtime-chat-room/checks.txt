#!/usr/bin/env python3
"""Compose a new main spec from an ADDED-only delta spec (new capability).

Usage: compose_main_spec.py <capability> <delta spec.md> <output spec.md>

Follows the openspec-sync-specs "Main Spec Format Reference": title
"# <capability> Specification", the delta's "## Purpose" body copied
verbatim, then a single "## Requirements" section holding the ADDED
requirement blocks verbatim. Refuses deltas with MODIFIED/REMOVED/RENAMED
sections (not allowed for a new capability).
"""
import re
import sys

cap, delta_path, out_path = sys.argv[1:4]
lines = open(delta_path, encoding="utf-8").read().split("\n")

sections = {}
current = None
for line in lines:
    m = re.match(r"^## (.+?)\s*$", line)
    if m:
        current = m.group(1)
        if current in sections:
            sys.exit(f"duplicate section {current!r} in {delta_path}")
        sections[current] = []
        continue
    if current is not None:
        sections[current].append(line)

allowed = {"Purpose", "ADDED Requirements"}
extra = set(sections) - allowed
if extra:
    sys.exit(f"unsupported sections for a new capability: {sorted(extra)}")
if "ADDED Requirements" not in sections:
    sys.exit("no ADDED Requirements")


def trim(block):
    while block and block[0].strip() == "":
        block = block[1:]
    while block and block[-1].strip() == "":
        block = block[:-1]
    return block


purpose = trim(sections.get("Purpose", []))
added = trim(sections["ADDED Requirements"])
if not purpose:
    sys.exit("no Purpose body (would need a TBD placeholder)")

out = [f"# {cap} Specification", "", "## Purpose", ""] + purpose + [
    "", "## Requirements", ""] + added
open(out_path, "w", encoding="utf-8").write("\n".join(out) + "\n")
print(f"wrote {out_path}: purpose {len(purpose)} line(s), requirements block {len(added)} line(s)")
[exit code: 0]

#!/usr/bin/env python3
"""Compare requirements/scenarios of a delta spec (ADDED section) with a main
spec (Requirements section). Written independently of compose_main_spec.py.

Usage: fidelity.py <delta spec.md> <main spec.md>
Exit 0 when every requirement and scenario (heading + exact body text, in
order) and the Purpose body match; exit 1 otherwise, listing differences.
"""
import re
import sys


def section(text, name):
    """Lines of the level-2 section `## <name>` up to the next level-2 heading."""
    out, inside = [], False
    for line in text.split("\n"):
        if re.match(r"^## ", line):
            inside = line.rstrip("\n") == f"## {name}"
            continue
        if inside:
            out.append(line)
    return out


def parse(lines):
    """-> list of (req_name, req_body, [(scen_name, scen_body)]), bodies exact
    except that blank lines at the edges of each block are dropped."""
    reqs = []
    cur_req = cur_scen = None
    for line in lines:
        m = re.match(r"^### Requirement: (.*)$", line)
        if m:
            cur_req = [m.group(1), [], []]
            reqs.append(cur_req)
            cur_scen = None
            continue
        m = re.match(r"^#### Scenario: (.*)$", line)
        if m:
            if cur_req is None:
                raise SystemExit("scenario before any requirement")
            cur_scen = [m.group(1), []]
            cur_req[2].append(cur_scen)
            continue
        if cur_scen is not None:
            cur_scen[1].append(line)
        elif cur_req is not None:
            cur_req[1].append(line)
        elif line.strip():
            raise SystemExit(f"text outside any requirement: {line!r}")
    edge = lambda b: "\n".join(b).strip("\n")
    return [(n, edge(b), [(sn, edge(sb)) for sn, sb in s]) for n, b, s in reqs]


delta_text = open(sys.argv[1], encoding="utf-8").read()
main_text = open(sys.argv[2], encoding="utf-8").read()
problems = []

for op in ("MODIFIED Requirements", "REMOVED Requirements", "RENAMED Requirements"):
    if f"\n## {op}" in "\n" + delta_text:
        problems.append(f"delta has unexpected section {op}")
for hdr in ("ADDED", "MODIFIED", "REMOVED", "RENAMED"):
    if re.search(rf"^## {hdr} Requirements", main_text, re.M):
        problems.append(f"main spec contains delta header {hdr}")

dp = "\n".join(section(delta_text, "Purpose")).strip("\n")
mp = "\n".join(section(main_text, "Purpose")).strip("\n")
if dp != mp:
    problems.append("Purpose body differs")

d = parse(section(delta_text, "ADDED Requirements"))
m = parse(section(main_text, "Requirements"))
dn, mn = [r[0] for r in d], [r[0] for r in m]
if dn != mn:
    problems.append(f"requirement list differs: delta={dn} main={mn}")
scen_total = 0
for (n, b, s), (n2, b2, s2) in zip(d, m):
    if b != b2:
        problems.append(f"requirement body differs: {n}")
    if [x[0] for x in s] != [x[0] for x in s2]:
        problems.append(f"scenario list differs in: {n}")
    for (sn, sb), (sn2, sb2) in zip(s, s2):
        if sn == sn2 and sb != sb2:
            problems.append(f"scenario body differs: {n} / {sn}")
    scen_total += len(s)

print(f"{sys.argv[2]}: requirements delta={len(d)} main={len(m)}; "
      f"scenarios delta={sum(len(r[2]) for r in d)} main={sum(len(r[2]) for r in m)}")
for p in problems:
    print("  MISMATCH:", p)
print("  RESULT:", "match" if not problems else "MISMATCH")
sys.exit(1 if problems else 0)
