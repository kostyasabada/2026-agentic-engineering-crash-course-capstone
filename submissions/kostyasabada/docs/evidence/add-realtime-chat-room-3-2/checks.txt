--- mutate.sh ---
#!/usr/bin/env bash
# mutate.sh <sed-expression> - applies one mutation to src/server/chat/chat.service.ts (run in the
# submission root), shows the diff, runs the service tests, restores the original from a backup
# copy, and verifies the restore by SHA-256. Exit: 0 if the mutation was killed and the restore
# verified, 1 otherwise.
F=src/server/chat/chat.service.ts
BK=$(mktemp)
cp "$F" "$BK"; before=$(sha256sum "$F" | cut -d" " -f1); echo "before: $before"
sed -i "$1" "$F"
if cmp -s "$F" "$BK"; then echo "mutation did not change the file"; cp "$BK" "$F"; rm -f "$BK"; exit 1; fi
diff -u "$BK" "$F" | tail -n +3
npx vitest run src/server/chat/chat.service.test.ts 2>&1 | grep -E "^ +×|Test Files|Tests  "
t=${PIPESTATUS[0]}; echo "vitest exit (mutant): $t"
cp "$BK" "$F"; rm -f "$BK"; after=$(sha256sum "$F" | cut -d" " -f1); echo "after restore: $after"
[ "$before" = "$after" ] && echo "restore verified" || { echo "RESTORE MISMATCH"; exit 1; }
[ "$t" -ne 0 ] && echo "mutant killed" || { echo "mutant SURVIVED"; exit 1; }

#!/usr/bin/env bash
# procs.sh - lists processes matching server.ts|next-server|next build|next dev|playwright|vitest
# whose working directory is inside the submission; prints nothing else about other processes.
ROOT=/home/ksabada/projects/AI_course/2026-agentic-engineering-crash-course-capstone/submissions/kostyasabada
n=0
for pid in $(pgrep -f 'server\.ts|next-server|next build|next dev|playwright|vitest'); do
  [ "$pid" = "$$" ] && continue
  cwd=$(readlink "/proc/$pid/cwd" 2>/dev/null) || continue
  case "$cwd" in "$ROOT"*) ps -o pid=,args= -p "$pid"; n=$((n+1));; esac
done
echo "matching processes in the submission: $n"
